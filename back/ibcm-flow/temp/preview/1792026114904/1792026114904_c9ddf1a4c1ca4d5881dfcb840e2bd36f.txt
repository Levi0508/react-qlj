import React, { useState, useEffect } from 'react';
import { Card, Table, Tabs, Input, Button, Select, Row, Col, message } from 'antd';
import { Icon } from 'antd';
import { PlusOutlined, DeleteOutlined } from '@ant-design/icons';
import services from '../../services';

const { TabPane } = Tabs;
const { Option } = Select;
const { TextArea } = Input;

function ReviewRecord({ clientId = clientId, curNode = {}, editable = false }) {
    const [activeTab, setActiveTab] = useState('0');
    const [showHistoryLogs, setShowHistoryLogs] = useState(true);
    const [latestData, setLatestData] = useState([]);
    const [historyData, setHistoryData] = useState([]);
    const [batchTotal, setBatchTotal] = useState(0);
    const [latestInfo, setLatestInfo] = useState({
        display_name: '',
        create_time: '',
        remark: ''
    });
    const [batchInfo, setBatchInfo] = useState({
        display_name: '',
        create_time: '',
        remark: ''
    });

    // 审核编辑相关状态
    const [tableData, setTableData] = useState([]); // 当前审核详情
    const [remark, setRemark] = useState(''); // 备注
    const [errorTypeOptions, setErrorTypeOptions] = useState([]); // 错误类型下拉框选项
    const [isEditableMode, setIsEditableMode] = useState(false); // 是否为可编辑模式
    const [initialLoadDone, setInitialLoadDone] = useState(false); // 标记是否已完成初次加载

    // 用于跟踪组件是否已挂载
    const isMounted = React.useRef(true);

    useEffect(() => {
        // 判断是否为可编辑模式：当 curNode.nodeCode === "hgrysh" 时
        const isEditMode = curNode?.nodeCode === 'hgrysh';
        setIsEditableMode(isEditMode);

        loadBatchTotal();
        loadApprovalHistory();
        // 加载初始数据
        const loadInitialData = async () => {
            try {
                // 加载批次总数
                const response = await services.selectBatchTotal({
                    clientId: clientId
                });

                if (response.success) {
                    const total = parseInt(response.data || '0', 10);
                    console.log('loadInitialData: 批次总数:', total);

                    if (isMounted.current) {
                        setBatchTotal(total);
                    }

                    // 批次总数加载完成后，如果是第一次加载且批次总数大于0，则加载第一个批次的数据
                    if (isMounted.current && !initialLoadDone && total > 0) {
                        console.log('初次加载，自动加载第一个tab页的数据');
                        // 直接调用loadHistoryDataByBatch而不是handleTabClick，避免重复调用loadApprListByBatch
                        await loadHistoryDataByBatch(0);
                        setInitialLoadDone(true);
                    }
                }
            } catch (error) {
                console.error('加载批次总数失败:', error);
            }
        };
        loadInitialData();

        if (isEditMode) {
            loadErrorTypeOptions();
            // 注册全局方法，用于获取审核数据
            window.getReviewRecordData = () => {
                return {
                    tableData: [...tableData],
                    remark: remark,
                    clientId: clientId
                };
            };
        }


        // 清理函数
        return () => {
            isMounted.current = false;
            if (window.getReviewRecordData) {
                delete window.getReviewRecordData;
            }
            if (window.errorTypeOptions) {
                delete window.errorTypeOptions;
            }
        };
    }, [clientId, curNode, editable, tableData, remark]); // 移除batchTotal和initialLoadDone依赖，避免无限循环

    // 加载审核批次总数
    const loadBatchTotal = async () => {
        try {
            const response = await services.selectBatchTotal({
                clientId: clientId
            });
            console.log('loadBatchTotal: 接口响应:', response);


            if (response.success) {
                // 接口返回的data是字符串类型的数字，需要转换为数字
                const total = parseInt(response.data || '0', 10);
                console.log('loadBatchTotal: 批次总数:', total);
                setBatchTotal(total);
            } else {
                console.error('获取批次总数失败:', response.msg);
                setBatchTotal(0);
            }
        } catch (error) {
            console.error('加载审核批次总数失败:', error);
            // 检查组件是否已挂载
            if (isMounted.current) {
                setBatchTotal(0);
            }
        }
    };



    // 加载审核记录数据
    const loadApprovalHistory = async () => {
        try {
            const response = await services.selectBatchTotal({
                clientId: clientId
            });

            if (response.success) {
                const total = parseInt(response.data || '0', 10);
                setBatchTotal(total);

            }
        } catch (error) {
            console.error('加载审核记录失败:', error);
        }
    };

    // 处理标签页点击
    const handleTabClick = async (tabKey) => {
        const tabIndex = parseInt(tabKey);
        console.log("tabIndex", tabIndex);
        await loadHistoryDataByBatch(tabIndex);
        // 同时加载该批次的审核列表信息
        await loadApprListByBatch(tabIndex);
    };

    // 加载历史审核数据
    const loadHistoryDataByBatch = async (tabIndex) => {
        try {
            // tabIndex 是从0开始的，批次号是从1开始的
            // 批次总数 = batchTotal
            // 历史批次编号 = 1, 2, ..., batchTotal
            // tabIndex 0 -> 批次 batchTotal-1
            // tabIndex 1 -> 批次 batchTotal-2

            const batchNum = batchTotal - tabIndex;

            // 确保批次号有效
            if (batchNum >= 1 && batchNum <= batchTotal) {
                const response = await services.selectApprDetails({
                    clientId: clientId,
                    batchNum: String(batchNum)
                });
                console.log("loadHistoryDataByBatch: 接口响应:", response)


                if (response.success && response.data) {
                    console.log("hahahahahahahahahahah");
                    let historyRecords = [];
                    try {
                        historyRecords = JSON.parse(response.data);
                    } catch (e) {
                        console.error('解析历史审核记录数据失败:', e);
                        historyRecords = response.data;
                    }

                    if (historyRecords && historyRecords.length > 0) {
                        loadHistoryData(historyRecords);
                    }
                }
            }
        } catch (error) {
            console.error('加载历史审核数据失败:', error);
        }
    };

    // 加载错误类型选项
    const loadErrorTypeOptions = async () => {
        try {
            // 调用接口获取错误类型
            const response = await services.selectErrType();

            // 检查组件是否已挂载
            if (!isMounted.current) {
                console.log('loadErrorTypeOptions: 组件已卸载，跳过状态更新');
                return;
            }

            if (response.success && response.data) {
                let errorTypes = [];
                try {
                    errorTypes = JSON.parse(response.data);
                } catch (e) {
                    console.error('解析错误类型数据失败:', e);
                    errorTypes = response.data;
                }
                if (errorTypes && errorTypes.length > 0) {
                    setErrorTypeOptions(errorTypes);
                } else {
                    setErrorTypeOptions([]);
                }
            } else {
                setErrorTypeOptions([]);
            }
        } catch (error) {
            console.error('加载错误类型失败:', error);
            // 检查组件是否已挂载
            if (isMounted.current) {
                // 出错时使用默认数据
                const defaultErrorTypes = [
                    { errType: '1', errDesc: '信息填写不完整' },
                    { errType: '2', errDesc: '附件缺失' },
                    { errType: '3', errDesc: '信息不一致' },
                    { errType: '4', errDesc: '资料不清晰' },
                    { errType: '5', errDesc: '其他问题' }
                ];
                setErrorTypeOptions(defaultErrorTypes);
            }
        }
    };

    // 添加行
    const addRow = () => {
        const newRow = {
            key: Date.now(),
            seq: tableData.length + 1,
            errorType: '',
            problemDesc: ''
        };
        setTableData([...tableData, newRow]);
    };

    // 移除行
    const removeRow = (index) => {
        const newData = [...tableData];
        newData.splice(index, 1);
        // 更新序号
        const updatedData = newData.map((item, idx) => ({
            ...item,
            seq: idx + 1
        }));
        setTableData(updatedData);
    };

    // 更新行数据
    const updateRow = (index, field, value) => {
        const newData = [...tableData];
        newData[index] = {
            ...newData[index],
            [field]: value
        };
        setTableData(newData);
    };

    // 提交审核
    const submitApproval = async () => {
        // 校验数据
        const errors = [];
        tableData.forEach((row, index) => {
            if (!row.errorType) {
                errors.push(`请选择第${index + 1}行的错误类型`);
            }
            if (!row.problemDesc) {
                errors.push(`请填写第${index + 1}行的问题描述`);
            }
        });

        if (errors.length > 0) {
            message.error(errors.join('\n'));
            return;
        }

        try {
            // 这里需要调用后端接口提交审核数据
            // 根据 base_appr.xwl 中的逻辑，需要调用 save_appr_list 和 save_appr_details 接口
            // 由于后端接口尚未开发，这里先模拟提交成功
            console.log('提交审核数据:', {
                clientId,
                tableData,
                remark
            });

            message.success('审核提交成功');

            // 清空表单数据
            setTableData([]);
            setRemark('');

            // 重新加载审核记录
            loadBatchTotal();
            // 重新加载当前批次的数据
            handleTabClick(activeTab);
        } catch (error) {
            console.error('提交审核失败:', error);
            message.error('提交审核失败');
        }
    };

    // 加载审核列表信息（获取备注、审核人、审核时间等）
    const loadApprListByBatch = async (tabIndex) => {
        try {
            // tabIndex 是从0开始的，批次号是从1开始的
            // 批次总数 = batchTotal
            // 历史批次编号 = 1, 2, ..., batchTotal
            // tabIndex 0 -> 批次 batchTotal-1
            // tabIndex 1 -> 批次 batchTotal-2
            const batchNum = batchTotal - tabIndex;

            // 确保批次号有效
            if (batchNum >= 1 && batchNum <= batchTotal) {
                // 注意：这里假设审核列表信息已经包含在审核详情中
                // 如果没有单独的selectApprList接口，我们可以从审核详情中提取
                // 或者如果后端有selectApprList接口，可以调用它
                console.log('loadApprListByBatch: 加载批次', batchNum, '的审核列表信息');

                // 这里可以调用selectApprList接口（如果存在）
                // const response = await services.selectApprList({
                //     clientId: clientId,
                //     batchNum: String(batchNum)
                // });

                // 暂时使用审核详情中的信息
                // 在实际开发中，应该调用selectApprList接口获取完整的审核列表信息
            }
        } catch (error) {
            console.error('加载审核列表信息失败:', error);
        }
    };

    // 设置历史数据
    const loadHistoryData = (batchData) => {

        const historyItems = Array.isArray(batchData) ? batchData : [batchData];
        console.log("historyItems", historyItems);
        setHistoryData(historyItems.map((item, index) => ({
            key: String(index + 1),
            disposeDesc: item.disposeDesc || '',
            auditDesc: item.auditDesc || '',
            createUser: item.createUser || '',
            createTime: item.createTime || ''
        })));

        // 设置批次信息 - 从第一条记录获取批次信息
        if (historyItems.length > 0) {
            const firstRecord = historyItems[0];
            setBatchInfo({
                display_name: firstRecord.createUser || '',
                create_time: firstRecord.createTime || '',
                remark: firstRecord.remark || firstRecord.auditRemark || '' // 尝试从不同字段获取备注
            });
        }
    };

    // 最新审核记录表格列
    const latestColumns = [
        {
            title: '序号',
            key: 'index',
            width: 180,
            render: (_, __, index) => index + 1
        },
        {
            title: '错误类型',
            dataIndex: 'auditDesc',
            width: 300
        },
        {
            title: '问题描述',
            dataIndex: 'disposeDesc',
        }
    ];

    // 历史审核记录表格列
    const historyColumns = [
        {
            title: '问题内容',
            dataIndex: 'disposeDesc',
        },
        {
            title: '问题分类',
            dataIndex: 'auditDesc',
            width: 300
        },
        {
            title: '提出人',
            dataIndex: 'createUser',
            width: 200
        },
        {
            title: '提出时间',
            dataIndex: 'createTime',
            width: 200
        }
    ];

    // 可编辑模式的表格列
    const editColumns = [
        {
            title: '序号',
            key: 'seq',
            width: 60,
            render: (_, __, index) => index + 1
        },
        {
            title: '错误类型',
            key: 'errorType',
            width: 250,
            render: (text, record, index) => (
                <Select
                    value={record.errorType}
                    onChange={(value) => updateRow(index, 'errorType', value)}
                    placeholder="请选择"
                    style={{ width: '100%' }}
                    disabled={!isEditableMode}
                >
                    {errorTypeOptions.map(option => (
                        <Option key={option.errType} value={option.errType}>
                            {option.errDesc}
                        </Option>
                    ))}
                </Select>
            )
        },
        {
            title: '问题描述',
            key: 'problemDesc',
            render: (text, record, index) => (
                <TextArea
                    value={record.problemDesc}
                    onChange={(e) => updateRow(index, 'problemDesc', e.target.value)}
                    placeholder="请输入问题描述"
                    autoSize={{ minRows: 1, maxRows: 3 }}
                    disabled={!isEditableMode}
                />
            )
        },
        {
            title: '操作',
            key: 'action',
            width: 100,
            render: (text, record, index) => (
                <Button
                    type="text"
                    size="small"
                    style={{ color: '#F56C6C' }}
                    onClick={() => removeRow(index)}
                    disabled={!isEditableMode}
                >
                    移除
                </Button>
            )
        }
    ];

    return (
        <Card className="box-card" id="approvalHistory" title="审核记录">
            {/* 可编辑模式：审核表单 */}
            <div>
                <div style={{ marginBottom: '20px' }}>
                    <h4 style={{ marginBottom: '16px', color: '#303133' }}>审核问题录入</h4>
                    <Table
                        dataSource={tableData}
                        columns={editColumns}
                        pagination={false}
                        bordered
                        size="small"
                        rowKey="key"
                    />
                    <div style={{ marginTop: '16px', textAlign: 'right' }}>
                        <Button
                            type="primary"
                            onClick={addRow}
                            icon={<PlusOutlined />}
                        >
                            添加行
                        </Button>
                    </div>
                </div>

                <div style={{ marginBottom: '20px' }}>
                    <div style={{ color: '#4b4b4b', padding: '20px 0 10px 0' }}>
                        备注
                    </div>
                    <TextArea
                        value={remark}
                        onChange={(e) => setRemark(e.target.value)}
                        placeholder="请输入备注"
                        rows={3}
                    />
                </div>
            </div>

            {/* 历史问题记录 */}
            <div>
                <div
                    style={{
                        color: '#4b4b4b',
                        padding: '20px 0 10px 0',
                        cursor: 'pointer',
                        userSelect: 'none'
                    }}
                    onClick={() => setShowHistoryLogs(!showHistoryLogs)}
                >
                    <span>
                        {showHistoryLogs ?
                            <Icon type="caret-down" /> :
                            <Icon type="caret-up" />
                        }
                    </span>
                    <span style={{ marginLeft: '8px' }}>历史问题记录</span>
                </div>

                {showHistoryLogs && (
                    <div>
                        {batchTotal === 0 ? (
                            <div style={{
                                color: '#8d8b8b',
                                height: '100%',
                                display: 'flex',
                                justifyContent: 'center',
                                alignItems: 'center'
                            }}>
                                暂无数据
                            </div>
                        ) : (
                            <Tabs activeKey={activeTab} onChange={handleTabClick} onClick={handleTabClick}>
                                {Array.from({ length: Math.max(0, batchTotal) }, (_, index) => (
                                    <TabPane
                                        key={String(index)}
                                        tab={`批次${batchTotal - index}`}>
                                        <Table
                                            dataSource={historyData}
                                            columns={historyColumns}
                                            pagination={false}
                                            bordered
                                            size="small"
                                        />
                                        <div style={{ marginTop: '16px' }}>
                                            <div style={{ color: '#4b4b4b', padding: '20px 0 10px 0' }}>
                                                备注
                                            </div>
                                            <Input.TextArea value={batchInfo.remark} disabled />
                                            <div style={{ color: '#4b4b4b', padding: '10px 0 0 0', textAlign: 'right' }}>
                                                审核人: {batchInfo.display_name}
                                            </div>
                                            <div style={{ color: '#4b4b4b', padding: '10px 0 0 0', textAlign: 'right' }}>
                                                审核时间: {batchInfo.create_time}
                                            </div>
                                        </div>
                                    </TabPane>
                                ))}
                            </Tabs>
                        )}
                    </div>
                )}
            </div>
        </Card>
    );
}

export default ReviewRecord;
